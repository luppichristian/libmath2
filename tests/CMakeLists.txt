cmake_minimum_required(VERSION 3.19)
project(libmath2-tests LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LM2_GTEST_FETCH "Fetch GoogleTest if not found" ON)

include(FetchContent)
enable_testing()
find_package(GTest 1.14.0 QUIET)
if(NOT GTest_FOUND AND LM2_GTEST_FETCH)
    FetchContent_Declare(googletest
        GIT_REPOSITORY https://github.com/google/googletest.git
        GIT_TAG v1.14.0)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_MakeAvailable(googletest)
endif()

add_library(libmath2 INTERFACE)
target_compile_features(libmath2 INTERFACE c_std_11)
target_include_directories(libmath2 INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
if(UNIX)
    target_link_libraries(libmath2 INTERFACE m)
endif()

file(GLOB_RECURSE LM2_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
list(FILTER LM2_TEST_SOURCES EXCLUDE REGEX "/consumer/")
add_executable(libmath2-tests ${LM2_TEST_SOURCES})
target_link_libraries(libmath2-tests PRIVATE libmath2 GTest::gtest_main)
include(GoogleTest)
gtest_discover_tests(libmath2-tests)

foreach(language IN ITEMS c cpp)
    if(language STREQUAL "c")
        set(consumer_source consumer/consumer.c)
    else()
        set(consumer_source consumer/consumer.cc)
    endif()
    set(configurations default no_extensions unprefixed no_extensions_unprefixed)
    if(language STREQUAL "cpp")
        list(APPEND configurations no_generics no_cpp_operators no_generics_unprefixed no_cpp_operators_unprefixed)
    endif()
    foreach(configuration IN LISTS configurations)
        set(consumer_target libmath2-consumer-${language}-${configuration})
        add_executable(${consumer_target} ${consumer_source})
        target_link_libraries(${consumer_target} PRIVATE libmath2)
        if(configuration MATCHES "^no_extensions" OR configuration MATCHES "^no_generics")
            target_compile_definitions(${consumer_target} PRIVATE LM2_NO_GENERICS)
        endif()
        if(configuration MATCHES "^no_extensions" OR configuration MATCHES "^no_cpp_operators")
            target_compile_definitions(${consumer_target} PRIVATE LM2_NO_CPP_OPERATORS)
        endif()
        if(configuration MATCHES "^no_generics")
            target_compile_definitions(${consumer_target} PRIVATE LM2_TEST_EXPECT_NO_GENERICS)
        elseif(configuration MATCHES "^no_cpp_operators")
            target_compile_definitions(${consumer_target} PRIVATE LM2_TEST_EXPECT_NO_OPERATORS)
        endif()
        if(configuration MATCHES "unprefixed$")
            target_compile_definitions(${consumer_target} PRIVATE LM2_ENABLE_UNPREFIXED_NAMES)
        endif()
        add_test(NAME ${consumer_target} COMMAND ${consumer_target})
    endforeach()
endforeach()

# A consumer can embed its own cute_c2 implementation in the same translation unit.
add_executable(libmath2-consumer-cute-isolation consumer/cute_isolation.c)
target_link_libraries(libmath2-consumer-cute-isolation PRIVATE libmath2)
add_test(NAME libmath2-consumer-cute-isolation COMMAND libmath2-consumer-cute-isolation)

add_executable(libmath2-consumer-unsafe consumer/unsafe.c)
target_compile_definitions(libmath2-consumer-unsafe PRIVATE LM2_UNSAFE)
target_link_libraries(libmath2-consumer-unsafe PRIVATE libmath2)
add_test(NAME libmath2-consumer-unsafe COMMAND libmath2-consumer-unsafe)

# Small modules should compile and run without pulling in unrelated modules.
foreach(language IN ITEMS c cpp)
    foreach(module IN ITEMS base safe vector2 circle edge3)
        set(target "libmath2-selective-${module}-${language}")
        add_executable(${target} "consumer/selective_${module}.${language}")
        target_link_libraries(${target} PRIVATE libmath2)
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
endforeach()

# Public headers must compile independently, with their implementations included.
file(GLOB_RECURSE LM2_PUBLIC_HEADERS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../include"
    "${CMAKE_CURRENT_SOURCE_DIR}/../include/lm2/*.h")
list(FILTER LM2_PUBLIC_HEADERS EXCLUDE REGEX "(_impl|lm2_c2_utils|lm2_c2_backend)\\.h$")
list(APPEND LM2_PUBLIC_HEADERS lm2.h)
foreach(language IN ITEMS c cpp)
    set(header_sources)
    foreach(header IN LISTS LM2_PUBLIC_HEADERS)
        string(REPLACE "/" "_" source_name "${header}")
        set(source "${CMAKE_CURRENT_BINARY_DIR}/header-checks/${language}/${source_name}.${language}")
        file(GENERATE OUTPUT "${source}" CONTENT "#include <${header}>\ntypedef int lm2_header_isolation_translation_unit;\n")
        list(APPEND header_sources "${source}")
    endforeach()
    add_library(libmath2-headers-${language} OBJECT ${header_sources})
    target_link_libraries(libmath2-headers-${language} PRIVATE libmath2)
endforeach()

# Separate translation units exercise the static-inline implementation contract.
foreach(language IN ITEMS c cpp)
    add_executable(libmath2-header-only-${language}
        consumer/header_only_${language}_main.${language}
        consumer/header_only_${language}_peer.${language})
    target_include_directories(libmath2-header-only-${language} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
    if(UNIX)
        target_link_libraries(libmath2-header-only-${language} PRIVATE m)
    endif()
    add_test(NAME libmath2-header-only-${language} COMMAND libmath2-header-only-${language})
endforeach()

# Cyclic geometry modules must work from either include order in separate TUs.
foreach(language IN ITEMS c cpp)
    add_executable(libmath2-geometry-order-${language}
        consumer/geometry_order_main.${language}
        consumer/geometry_order_peer.${language})
    target_link_libraries(libmath2-geometry-order-${language} PRIVATE libmath2)
    add_test(NAME libmath2-geometry-order-${language} COMMAND libmath2-geometry-order-${language})
endforeach()
